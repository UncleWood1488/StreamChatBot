import aiohttp
from platforms.base import BasePlatformAdapter
from config import Config

TWITCH_API_BASE = "https://api.twitch.tv/helix"

class TwitchAdapter(BasePlatformAdapter):
    def __init__(self):
        self.broadcaster_id = Config.TWITCH_BROADCASTER_ID
        self.headers = {
            "Client-ID": Config.TWITCH_CLIENT_ID,
            "Authorization": f"Bearer {Config.TWITCH_TOKEN}",
            "Content-Type": "application/json",
        }
        self._session: aiohttp.ClientSession | None = None

    @property
    def name(self) -> str:
        return "Twitch"

    async def _get_session(self) -> aiohttp.ClientSession:
        if self._session is None or self._session.closed:
            self._session = aiohttp.ClientSession()
        return self._session

    async def _patch_channel(self, payload: dict) -> bool:
        url = f"{TWITCH_API_BASE}/channels?broadcaster_id={self.broadcaster_id}"
        session = await self._get_session()
        async with session.patch(url, headers=self.headers, json=payload) as resp:
            if resp.status == 204:
                return True
            text = await resp.text()
            raise RuntimeError(f"Twitch API вернул {resp.status}: {text}")

    async def update_title(self, title: str) -> bool:
        return await self._patch_channel({"title": title[:140]})

    async def update_category(self, category_name: str) -> bool:
        game_id = await self._find_game_id(category_name)
        if not game_id:
            raise ValueError(f'Игра "{category_name}" не найдена на Twitch.')
        return await self._patch_channel({"game_id": game_id})

    async def _find_game_id(self, name: str) -> str | None:
        url = f"{TWITCH_API_BASE}/games?name={name}"
        session = await self._get_session()
        async with session.get(url, headers=self.headers) as resp:
            resp.raise_for_status()
            data = await resp.json()
            if data["data"]:
                return data["data"][0]["id"]
        return None

    async def close(self):
        if self._session and not self._session.closed:
            await self._session.close()
